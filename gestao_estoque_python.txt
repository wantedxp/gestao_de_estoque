import React, { useState } from 'react';
import { View, Text, TextInput, Button, FlatList, StyleSheet } from 'react-native';

const VendaApp = () => {
  const [produto, setProduto] = useState('');
  const [preco, setPreco] = useState('');
  const [quantidade, setQuantidade] = useState('');
  const [vendas, setVendas] = useState([]);

  const adicionarVenda = () => {
    if (!produto || !preco || !quantidade) {
      alert("Todos os campos são obrigatórios!");
      return;
    }
    const total = parseFloat(preco) * parseInt(quantidade);
    setVendas([...vendas, { produto, preco, quantidade, total }]);
    setProduto('');
    setPreco('');
    setQuantidade('');
  };

  const calcularTotalGeral = () => {
    return vendas.reduce((total, venda) => total + venda.total, 0);
  };

  return (
    <View style={styles.container}>
      <Text style={styles.title}>Gestão de Vendas</Text>

      <TextInput
        style={styles.input}
        placeholder="Nome do produto"
        placeholderTextColor="#ccc"
        value={produto}
        onChangeText={setProduto}
      />
      <TextInput
        style={styles.input}
        placeholder="Preço unitário"
        keyboardType="numeric"
        placeholderTextColor="#ccc"
        value={preco}
        onChangeText={setPreco}
      />
      <TextInput
        style={styles.input}
        placeholder="Quantidade"
        keyboardType="numeric"
        placeholderTextColor="#ccc"
        value={quantidade}
        onChangeText={setQuantidade}
      />

      <Button title="Adicionar Venda" onPress={adicionarVenda} color="#4CAF50" />

      <FlatList
        data={vendas}
        keyExtractor={(item, index) => index.toString()}
        renderItem={({ item }) => (
          <View style={styles.vendaItem}>
            <Text style={styles.itemText}>{item.produto}</Text>
            <Text style={styles.itemText}>Preço: R${item.preco}</Text>
            <Text style={styles.itemText}>Quantidade: {item.quantidade}</Text>
            <Text style={styles.itemText}>Total: R${item.total.toFixed(2)}</Text>
          </View>
        )}
      />

      <Text style={styles.total}>Total Geral: R${calcularTotalGeral().toFixed(2)}</Text>
    </View>
  );
};

const styles = StyleSheet.create({
  container: {
    flex: 1,
    padding: 20,
    justifyContent: 'flex-start',
    backgroundColor: '#121212',  // Fundo preto
  },
  title: {
    fontSize: 24,
    fontWeight: 'bold',
    textAlign: 'center',
    marginBottom: 20,
    color: '#fff',  // Texto branco
  },
  input: {
    height: 40,
    borderColor: '#555',
    borderWidth: 1,
    marginBottom: 10,
    paddingLeft: 8,
    borderRadius: 5,
    color: '#fff',  // Texto dos inputs branco
  },
  vendaItem: {
    padding: 10,
    marginBottom: 10,
    backgroundColor: '#333',  // Fundo escuro para os itens
    borderRadius: 5,
    borderColor: '#444',
    borderWidth: 1,
  },
  itemText: {
    color: '#fff',  // Texto dos itens branco
  },
  total: {
    fontSize: 20,
    fontWeight: 'bold',
    textAlign: 'center',
    marginTop: 20,
    color: '#fff',  // Total branco
  },
});

export default VendaApp;
